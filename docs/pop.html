Ahora, como estimar una poblacion para cada año? Se requiere una funcion de interpolacion lineal o spline para cada provincia. y una serie con los años de destino de 1914-2022

serie_tiempo <- seq(1914, 2022, by=1)

creamos una funcion

interpol <- function(data) {
  #Lineal
  interp_lineal <- approx(data$ronda_censal, data$poblacion, xout = serie_tiempo)$y
  #Spline
  interp_spline <- spline(data$ronda_censal, data$poblacion, xout = serie_tiempo)$y
  
  #creacion de dataframe
  
  pop_intepol <- data.frame(
    Year = serie_tiempo,
    estimada_lineal = interp_lineal,
    estimada_spline = interp_spline
  )
  
  return(pop_intepol)
}

Ahora aplicamos esta funcion al dataframe de origen de los datos.

poblacion_interp <- poblacion_censos |> 
  group_by(Provincia) |> 
  group_modify(~interpol(.x))

grafico5 <- poblacion_interp |> 
  group_by(Year) |> 
  summarise(
    poblacion_total_l = sum(estimada_lineal),
    poblacion_total_sp = sum(estimada_spline)
  ) |> 
  ggplot()+
  geom_line(aes(x=Year, y=poblacion_total_l, colour = "blue"))+
